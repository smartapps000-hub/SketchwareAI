    /**
     * Downloads the files a page offers with Android's download manager: into Downloads/folder (Android 10+, or with
     * storage permission), else the app's own Downloads folder. mimeFilter: empty, a star or the any-type pattern for
     * every file; otherwise MIME types or file endings separated by commas. A long press on an image offers to
     * download it too.
     */
    public static void allowDownloads(WebView webView, String folder, boolean notify, String mimeFilter) {
        if (webView == null) return;
        State state = state(webView);
        state.downloads = true;
        state.folder = folder == null ? "" : folder.trim();
        state.notify = notify;
        state.mimeFilter = mimeFilter == null ? "" : mimeFilter.trim();
        listen(webView, state);
        imageLongPress(webView, state);
    }

    public static void setDownloadEvents(WebView webView, DownloadEvents events) {
        if (webView == null) return;
        State state = state(webView);
        state.events = events;
        listen(webView, state);
    }

    private static void listen(final WebView webView, final State state) {
        if (state.listening) return;
        state.listening = true;
        webView.setDownloadListener(new DownloadListener() {
            @Override
            public void onDownloadStart(String url, String userAgent, String contentDisposition, String mimeType, long contentLength) {
                download(webView, state, url, userAgent, contentDisposition, mimeType, contentLength);
            }
        });
    }

    /** A download the page (or a window it opened) starts: the event, then the download manager when allowed. */
    private static void download(WebView webView, State state, String url, String userAgent, String contentDisposition,
                                 String mimeType, long contentLength) {
        String fileName = fileName(url, contentDisposition, mimeType);
        String type = mimeType(mimeType, fileName);
        if (state.events != null) state.events.onDownloadStart(url, type, (double) contentLength);
        if (state.downloads && (mimeAllowed(type, fileName, state.mimeFilter)
                || mimeType != null && mimeAllowed(mimeType, fileName, state.mimeFilter))) {
            enqueue(webView.getContext(), webView, state, url, userAgent, fileName, type, state.folder, state.notify);
        }
    }

    /**
     * Downloads a file from a link with Android's download manager, without a WebView. fileName: empty to take the
     * server's name; folder: inside Downloads.
     */
    public static void downloadFile(Object host, String url, String folder, String fileName, boolean notify) {
        downloadFile(host, url, folder, fileName, notify, "");
    }

    /**
     * Downloads a file from a link with Android's download manager, without a WebView. fileName: empty to take the
     * server's name; folder: inside Downloads. typeFilter (v1.9.7): empty, a star or "Default" = Chrome-grade naming;
     * a category ("audio", "video", "images", "packages", "documents") or a custom ending / MIME (".flac",
     * "application/json") only gives a name that has no ending the category's first ending. It never blocks the
     * download (this block is an explicit action).
     */
    public static void downloadFile(Object host, String url, String folder, String fileName, boolean notify, String typeFilter) {
        Context context = context(host);
        if (context == null || url == null || url.trim().length() == 0) return;
        url = url.trim();
        String name = fileName == null ? null : safeName(fileName.trim());
        if (name == null) name = fileName(url, null, null);
        String extensions = categoryExtensions(typeFilter);
        if (!hasExtension(name) && extensions.length() > 0) {
            String ending = firstEnding(extensions);
            if (ending != null) name = name + "." + ending;
        }
        enqueue(context, null, null, url, null, name, mimeType(null, name), folder == null ? "" : folder.trim(), notify);
    }

    /** The comma list of endings a download-type token stands for; a custom value is returned as it is. */
    public static String categoryExtensions(String filter) {
        if (filter == null) return "";
        String f = filter.trim().toLowerCase(Locale.US);
        if (f.length() == 0 || f.equals("*") || f.equals("*/*") || f.startsWith("default")) return "";
        if (f.equals("audio")) return "mp3,m4a,wav,flac,ogg,aac";
        if (f.equals("video")) return "mp4,mkv,webm,avi,mov";
        if (f.equals("image") || f.equals("images")) return "png,jpg,jpeg,webp,gif";
        if (f.equals("package") || f.equals("packages")) return "apk,aab,zip,rar,7z";
        if (f.equals("document") || f.equals("documents")) return "pdf,docx,doc,txt";
        return filter.trim();
    }

    private static String firstEnding(String extensions) {
        String[] parts = extensions.split(",");
        for (int i = 0; i < parts.length; i++) {
            String part = parts[i].trim();
            if (part.startsWith(".")) part = part.substring(1);
            if (part.length() > 0 && part.indexOf('/') < 0) return part.toLowerCase(Locale.US);
        }
        return null;
    }

    /** Whether a file of this MIME type passes the filter (comma-separated, type/* allowed, case ignored). */
    public static boolean mimeAllowed(String mimeType, String filter) {
        return mimeAllowed(mimeType, null, filter);
    }

    /**
     * Whether a file passes the filter: an empty filter, a star or the any-type pattern = every file (like a
     * browser); otherwise MIME types (application/pdf, audio/ with a star) or file endings (.mp3, mp3, .apk),
     * separated by commas, case ignored.
     */
    public static boolean mimeAllowed(String mimeType, String fileName, String filter) {
        if (filter == null || filter.trim().length() == 0) return true;
        String name = fileName == null ? "" : fileName.toLowerCase(Locale.US);
        String type = mimeType == null ? "" : mimeType.trim().toLowerCase(Locale.US);
        int semicolon = type.indexOf(';');
        if (semicolon >= 0) type = type.substring(0, semicolon).trim();
        String[] allowed = filter.toLowerCase(Locale.US).split(",");
        for (int i = 0; i < allowed.length; i++) {
            String a = allowed[i].trim();
            if (a.length() == 0) continue;
            if (a.equals("*") || a.equals("*/*") || a.equals(type)) return true;
            if (a.indexOf('/') < 0) {
                // a file ending: .mp3 or mp3
                String ending = a.startsWith(".") ? a : "." + a;
                if (name.endsWith(ending)) return true;
                String byEnding = MimeTypeMap.getSingleton().getMimeTypeFromExtension(ending.substring(1));
                if (byEnding != null && byEnding.equalsIgnoreCase(type)) return true;
                continue;
            }
            if (a.endsWith("/*") && type.startsWith(a.substring(0, a.length() - 1))) return true;
        }
        return false;
    }

    /**
     * The file name, in three stages: the server's Content-Disposition (filename*= with its charset first, then
     * filename=); else a name in the link (a query parameter such as file= or name= that has an ending, else the
     * path's last part); then, without an ending, the one the MIME type stands for. Never ".bin": a name still
     * without an ending gets it from the file's first bytes when it is downloaded.
     */
    public static String fileName(String url, String contentDisposition, String mimeType) {
        String name = dispositionName(contentDisposition);
        if (name == null) name = urlName(url);
        if (name == null) name = "download";
        return withMimeExtension(name, mimeType);
    }

    /** The name with the ending its MIME type stands for when it has none (never "bin": octet-stream says nothing). */
    public static String withMimeExtension(String name, String mimeType) {
        String type = baseType(mimeType);
        if (scriptEnding(name) && type.length() > 0 && !type.equals("text/html") && !type.equals("application/octet-stream")
                && !type.equals("binary/octet-stream")) {
            String ending = MimeTypeMap.getSingleton().getExtensionFromMimeType(type);
            if (ending != null && !ending.equals("bin")) return name.substring(0, name.lastIndexOf('.')) + "." + ending;
        }
        if (hasExtension(name) && !(unknownEnding(name) && mediaType(type))) return name;
        if (type.length() > 0 && !type.equals("application/octet-stream") && !type.equals("binary/octet-stream")) {
            String ending = MimeTypeMap.getSingleton().getExtensionFromMimeType(type);
            if (ending != null && !ending.equals("bin")) return name + "." + ending;
        }
        return name;
    }

    /** The name in a Content-Disposition header, or null. */
    public static String dispositionName(String contentDisposition) {
        if (contentDisposition == null) return null;
        String name = null;
        String[] parts = contentDisposition.split(";");
        for (int i = 0; i < parts.length && name == null; i++) {
            String part = parts[i].trim();
            if (part.toLowerCase(Locale.US).startsWith("filename*=")) {
                String value = unquote(part.substring(10).trim());
                String charset = "UTF-8";
                int quote = value.indexOf("''");
                if (quote >= 0) {
                    if (value.substring(0, quote).trim().equalsIgnoreCase("ISO-8859-1")) charset = "ISO-8859-1";
                    value = value.substring(quote + 2);
                }
                try {
                    name = java.net.URLDecoder.decode(value.replace("+", "%2B"), charset);
                } catch (Exception e) {
                    name = value;
                }
            }
        }
        for (int i = 0; i < parts.length && name == null; i++) {
            String part = parts[i].trim();
            if (part.toLowerCase(Locale.US).startsWith("filename=")) {
                name = unquote(part.substring(9).trim());
            }
        }
        return name == null ? null : safeName(name);
    }

    /** Endings a server script has in its link (download.php): the file it sends is something else. */
    private static final String[] SCRIPT_ENDINGS = {"php", "asp", "aspx", "jsp", "cgi", "pl", "do", "action", "htm", "html"};

    private static boolean unknownEnding(String name) {
        int dot = name.lastIndexOf('.');
        return dot > 0 && MimeTypeMap.getSingleton().getMimeTypeFromExtension(name.substring(dot + 1).toLowerCase(Locale.US)) == null;
    }

    private static boolean mediaType(String type) {
        return type.startsWith("audio/") || type.startsWith("video/") || type.startsWith("image/") || type.equals("application/pdf");
    }

    private static boolean scriptEnding(String name) {
        int dot = name == null ? -1 : name.lastIndexOf('.');
        if (dot < 0) return false;
        String ending = name.substring(dot + 1).toLowerCase(Locale.US);
        for (int i = 0; i < SCRIPT_ENDINGS.length; i++) {
            if (SCRIPT_ENDINGS[i].equals(ending)) return true;
        }
        return false;
    }

    private static final String[] NAME_PARAMETERS = {"filename", "file_name", "file", "name", "title", "fn", "f",
            "download", "dl", "attachment", "media", "song", "video"};

    /** A file name in a link: a query parameter that names a file with an ending, else the path's last part. */
    public static String urlName(String url) {
        if (url == null) return null;
        String link = url;
        int hash = link.indexOf('#');
        if (hash >= 0) link = link.substring(0, hash);
        int question = link.indexOf('?');
        String query = question >= 0 ? link.substring(question + 1) : "";
        String path = question >= 0 ? link.substring(0, question) : link;
        if (query.length() > 0) {
            String[] pairs = query.split("&");
            for (int k = 0; k < NAME_PARAMETERS.length; k++) {
                for (int i = 0; i < pairs.length; i++) {
                    int equals = pairs[i].indexOf('=');
                    if (equals <= 0) continue;
                    if (!pairs[i].substring(0, equals).equalsIgnoreCase(NAME_PARAMETERS[k])) continue;
                    String value = decodeQuery(pairs[i].substring(equals + 1));
                    int slash = Math.max(value.lastIndexOf('/'), value.lastIndexOf('\\'));
                    if (slash >= 0) value = value.substring(slash + 1);
                    if (hasExtension(value)) return safeName(value);
                }
            }
        }
        int scheme = path.indexOf("://");
        if (scheme >= 0) {
            int slash = path.indexOf('/', scheme + 3);
            path = slash >= 0 ? path.substring(slash) : "";
        }
        int semicolon = path.indexOf(';');
        if (semicolon >= 0) path = path.substring(0, semicolon);
        while (path.endsWith("/")) path = path.substring(0, path.length() - 1);
        String last = path.substring(path.lastIndexOf('/') + 1);
        last = decode(last);
        return last.trim().length() == 0 ? null : safeName(last.trim());
    }

    private static String decodeQuery(String value) {
        // form encoding: + is a space
        try {
            return java.net.URLDecoder.decode(value, "UTF-8");
        } catch (Exception e) {
            return value;
        }
    }

    private static String decode(String value) {
        try {
            return java.net.URLDecoder.decode(value.replace("+", "%2B"), "UTF-8");
        } catch (Exception e) {
            return value;
        }
    }

    /**
     * Whether a name ends in a file ending: one Android knows (mp3, torrent ...), or 2 to 8 letters and digits with a
     * letter, all lower or all upper case (xapk, MKV). "Mr.Bean", "Still D.R.E" and "Invoice 12.2026" have none.
     */
    public static boolean hasExtension(String name) {
        if (name == null) return false;
        int dot = name.lastIndexOf('.');
        if (dot <= 0 || dot == name.length() - 1) return false;
        String ending = name.substring(dot + 1);
        if (MimeTypeMap.getSingleton().getMimeTypeFromExtension(ending.toLowerCase(Locale.US)) != null) return true;
        if (ending.length() < 2 || ending.length() > 8) return false;
        boolean letter = false;
        for (int i = 0; i < ending.length(); i++) {
            char c = ending.charAt(i);
            if (c > 127 || !Character.isLetterOrDigit(c)) return false;
            if (Character.isLetter(c)) letter = true;
        }
        return letter && (ending.equals(ending.toLowerCase(Locale.US)) || ending.equals(ending.toUpperCase(Locale.US)));
    }

    /**
     * A name for the download folder: no folders and no characters a path or a Uri reads (% # ? : * " < > |, control
     * characters), no leading dots, at most 150 characters (the ending kept); null when nothing is left.
     */
    private static String safeName(String name) {
        StringBuilder clean = new StringBuilder();
        for (int i = 0; i < name.length(); i++) {
            char c = name.charAt(i);
            if (c < 32 || c == 127 || "/\\:*?\"<>|%#".indexOf(c) >= 0) {
                clean.append('_');
            } else {
                clean.append(c);
            }
        }
        String result = clean.toString().trim();
        while (result.startsWith(".")) result = result.substring(1).trim();
        if (result.length() == 0) return null;
        if (result.length() > 150) {
            int dot = result.lastIndexOf('.');
            String ending = dot > 0 && result.length() - dot <= 10 ? result.substring(dot) : "";
            int cut = 150 - ending.length();
            if (Character.isHighSurrogate(result.charAt(cut - 1))) cut--;
            result = result.substring(0, cut) + ending;
        }
        return result;
    }

    private static String unquote(String value) {
        if (value.length() >= 2 && value.startsWith("\"") && value.endsWith("\"")) {
            return value.substring(1, value.length() - 1);
        }
        return value;
    }

    private static String baseType(String mimeType) {
        String type = mimeType == null ? "" : mimeType.trim().toLowerCase(Locale.US);
        int semicolon = type.indexOf(';');
        return semicolon >= 0 ? type.substring(0, semicolon).trim() : type;
    }

    /** The MIME type the server sent, or the one the file name's ending stands for when it sent none. */
    public static String mimeType(String mimeType, String fileName) {
        String type = mimeType == null ? "" : mimeType.trim();
        if (type.length() > 0 && !type.equalsIgnoreCase("application/octet-stream")) return type;
        int dot = fileName == null ? -1 : fileName.lastIndexOf('.');
        if (dot >= 0) {
            String byEnding = MimeTypeMap.getSingleton().getMimeTypeFromExtension(fileName.substring(dot + 1).toLowerCase(Locale.US));
            if (byEnding != null) return byEnding;
        }
        return type.length() > 0 ? type : "application/octet-stream";
    }

    /** The file ending a file's first bytes show (mp3, mp4, m4a, pdf, zip, png, jpg ...), or null. */
    public static String extensionFromBytes(byte[] b, int n) {
        if (b == null || n < 4) return null;
        if (b[0] == 'I' && b[1] == 'D' && b[2] == '3') return "mp3";
        if ((b[0] & 0xFF) == 0xFF && ((b[1] & 0xE0) == 0xE0) && ((b[1] & 0x06) != 0)) {
            // an MPEG audio frame (layer bits set); AAC ADTS has the layer bits 0
            return "mp3";
        }
        if ((b[0] & 0xFF) == 0xFF && ((b[1] & 0xF6) == 0xF0)) return "aac";
        if (b[0] == '%' && b[1] == 'P' && b[2] == 'D' && b[3] == 'F') return "pdf";
        if (b[0] == 'P' && b[1] == 'K' && (b[2] == 3 || b[2] == 5 || b[2] == 7)) return "zip";
        if ((b[0] & 0xFF) == 0x89 && b[1] == 'P' && b[2] == 'N' && b[3] == 'G') return "png";
        if ((b[0] & 0xFF) == 0xFF && (b[1] & 0xFF) == 0xD8 && (b[2] & 0xFF) == 0xFF) return "jpg";
        if (b[0] == 'G' && b[1] == 'I' && b[2] == 'F' && b[3] == '8') return "gif";
        if (b[0] == 'O' && b[1] == 'g' && b[2] == 'g' && b[3] == 'S') return "ogg";
        if (b[0] == 'f' && b[1] == 'L' && b[2] == 'a' && b[3] == 'C') return "flac";
        if ((b[0] & 0xFF) == 0x1A && (b[1] & 0xFF) == 0x45 && (b[2] & 0xFF) == 0xDF && (b[3] & 0xFF) == 0xA3) return "webm";
        if (n >= 12 && b[0] == 'R' && b[1] == 'I' && b[2] == 'F' && b[3] == 'F') {
            if (b[8] == 'W' && b[9] == 'A' && b[10] == 'V' && b[11] == 'E') return "wav";
            if (b[8] == 'W' && b[9] == 'E' && b[10] == 'B' && b[11] == 'P') return "webp";
            if (b[8] == 'A' && b[9] == 'V' && b[10] == 'I') return "avi";
        }
        if (n >= 12 && b[4] == 'f' && b[5] == 't' && b[6] == 'y' && b[7] == 'p') {
            String brand = new String(b, 8, 4);
            if (brand.startsWith("M4A")) return "m4a";
            if (brand.startsWith("qt")) return "mov";
            if (brand.startsWith("3g")) return "3gp";
            return "mp4";
        }
        String text = new String(b, 0, Math.min(n, 64)).trim().toLowerCase(Locale.US);
        if (text.startsWith("<svg") || text.startsWith("<?xml") && text.contains("<svg")) return "svg";
        if (text.startsWith("<!doctype html") || text.startsWith("<html")) return "html";
        return null;
    }

    private static void enqueue(final Context context, final WebView webView, final State state, final String url,
                                final String userAgent, final String fileName, final String mimeType, final String folder,
                                final boolean notify) {
        final Context app = context.getApplicationContext();
        String cookieHeader = null;
        try {
            cookieHeader = CookieManager.getInstance().getCookie(url);
        } catch (Exception ignored) {
            // no WebView installed on the phone
        }
        final String cookies = cookieHeader;
        register(app);
        // asked off the screen's thread: the page never waits, and a name without an ending can be checked online
        new Thread(new Runnable() {
            @Override
            public void run() {
                String name = fileName;
                String type = mimeType;
                try {
                    if ((!hasExtension(name) || scriptEnding(name)) && (url.startsWith("http://") || url.startsWith("https://"))) {
                        String[] probe = probe(url, cookies, userAgent);
                        if (probe[0] != null) {
                            name = probe[0];
                        } else if (probe[1] != null) {
                            name = withMimeExtension(scriptEnding(name) ? name.substring(0, name.lastIndexOf('.')) : name, probe[1]);
                        }
                        type = mimeType(probe[1] != null ? probe[1] : type, name);
                    }
                    DownloadManager.Request request = new DownloadManager.Request(Uri.parse(url));
                    if (type != null && type.length() > 0) request.setMimeType(type);
                    if (cookies != null) request.addRequestHeader("Cookie", cookies);
                    if (userAgent != null) request.addRequestHeader("User-Agent", userAgent);
                    request.setTitle(name);
                    request.setNotificationVisibility(notify
                            ? DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED
                            : DownloadManager.Request.VISIBILITY_VISIBLE);
                    String path = folder.length() == 0 ? name : folder + "/" + name;
                    boolean shared = Build.VERSION.SDK_INT >= 29 || app.checkPermission(
                            "android.permission.WRITE_EXTERNAL_STORAGE", android.os.Process.myPid(), android.os.Process.myUid())
                            == PackageManager.PERMISSION_GRANTED;
                    if (shared) {
                        request.setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, path);
                        if (Build.VERSION.SDK_INT < 29) allowScanning(request);
                    } else {
                        request.setDestinationInExternalFilesDir(app, Environment.DIRECTORY_DOWNLOADS, path);
                    }
                    DownloadManager manager = (DownloadManager) app.getSystemService(Context.DOWNLOAD_SERVICE);
                    synchronized (pending) {
                        // under the receiver's lock: a download that ends at once still finds its entry
                        long id = manager.enqueue(request);
                        pending.put(id, new Object[]{webView == null ? null : new WeakReference<WebView>(webView), url, name});
                    }
                } catch (Exception e) {
                    // blob: and data: links, a missing download manager, a refused folder
                    final String failed = name;
                    new Handler(Looper.getMainLooper()).post(new Runnable() {
                        @Override
                        public void run() {
                            if (state != null && state.events != null) state.events.onDownloadComplete(url, failed, false);
                        }
                    });
                }
            }
        }).start();
    }

    @SuppressWarnings("deprecation")
    private static void allowScanning(DownloadManager.Request request) {
        request.allowScanningByMediaScanner();
    }

    /**
     * Asks the server about a link whose name has no ending, with a HEAD request only (owner's v1.9.7 choice: a GET
     * could spend a one-time download link): {name from its Content-Disposition, its MIME type}, each null when
     * unknown. Redirects are followed here, and the cookies go only to the link's own host and protocol. The file's
     * own bytes are read after the download instead (renameBySniff), which spends nothing.
     */
    private static String[] probe(String url, String cookies, String userAgent) {
        String[] result = new String[2];
        HttpURLConnection connection = null;
        try {
            URL target = new URL(url);
            String host = target.getHost();
            String protocol = target.getProtocol();
            for (int hop = 0; hop < 6; hop++) {
                connection = (HttpURLConnection) target.openConnection();
                connection.setConnectTimeout(4000);
                connection.setReadTimeout(4000);
                connection.setInstanceFollowRedirects(false);
                connection.setRequestMethod("HEAD");
                if (cookies != null && host.equalsIgnoreCase(target.getHost()) && protocol.equals(target.getProtocol())) {
                    connection.setRequestProperty("Cookie", cookies);
                }
                if (userAgent != null) connection.setRequestProperty("User-Agent", userAgent);
                int code = connection.getResponseCode();
                String location = connection.getHeaderField("Location");
                if (code >= 300 && code < 400 && location != null) {
                    connection.disconnect();
                    connection = null;
                    target = new URL(target, location);
                    if (!"http".equals(target.getProtocol()) && !"https".equals(target.getProtocol())) return result;
                    continue;
                }
                if (code < 200 || code >= 300) return result;
                String name = dispositionName(connection.getHeaderField("Content-Disposition"));
                if (name != null && hasExtension(name)) result[0] = name;
                String type = baseType(connection.getContentType());
                if (type.length() > 0 && !type.equals("application/octet-stream") && !type.equals("binary/octet-stream")
                        && !type.equals("text/html")) {
                    result[1] = type;
                }
                return result;
            }
        } catch (Exception ignored) {
            // no answer: the name stays as it is
        } finally {
            if (connection != null) connection.disconnect();
        }
        return result;
    }

    /**
     * After a download, when the saved file still has no ending and it is a real file we can read (the app's own
     * folder, or the public folder before Android 10), reads its first bytes and renames it with the ending they show.
     * This spends no network request. Returns the final name.
     */
    private static String renameBySniff(String localUri, String name) {
        if (localUri == null || hasExtension(name)) return name;
        if (!localUri.startsWith("file://")) return name;
        try {
            java.io.File file = new java.io.File(Uri.parse(localUri).getPath());
            if (!file.isFile()) return name;
            byte[] first = new byte[64];
            int read;
            java.io.FileInputStream in = new java.io.FileInputStream(file);
            try {
                read = in.read(first);
            } finally {
                in.close();
            }
            String ending = extensionFromBytes(first, read);
            if (ending == null) return name;
            java.io.File renamed = new java.io.File(file.getParentFile(), name + "." + ending);
            if (!renamed.exists() && file.renameTo(renamed)) return name + "." + ending;
        } catch (Exception ignored) {
            // a content:// file, no read permission: keep the name
        }
        return name;
    }

    private static void register(Context context) {
        if (receiver != null) return;
        receiver = new BroadcastReceiver() {
            @Override
            public void onReceive(Context c, Intent intent) {
                long id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1);
                synchronized (pending) {
                    if (!pending.containsKey(id)) return;
                }
                // the broadcast can come from any app: only the download manager's own final status ends an entry
                int status = -1;
                String localUri = null;
                Cursor cursor = null;
                try {
                    DownloadManager manager = (DownloadManager) c.getSystemService(Context.DOWNLOAD_SERVICE);
                    cursor = manager.query(new DownloadManager.Query().setFilterById(id));
                    if (cursor != null && cursor.moveToFirst()) {
                        int column = cursor.getColumnIndex(DownloadManager.COLUMN_STATUS);
                        if (column >= 0) status = cursor.getInt(column);
                        int uriColumn = cursor.getColumnIndex(DownloadManager.COLUMN_LOCAL_URI);
                        if (uriColumn >= 0) localUri = cursor.getString(uriColumn);
                    } else {
                        // removed from the download manager (cancelled in the notification)
                        status = DownloadManager.STATUS_FAILED;
                    }
                } catch (Exception ignored) {
                } finally {
                    if (cursor != null) cursor.close();
                }
                if (status == -1) status = DownloadManager.STATUS_FAILED;
                if (status != DownloadManager.STATUS_SUCCESSFUL && status != DownloadManager.STATUS_FAILED) return;
                Object[] entry;
                synchronized (pending) {
                    entry = pending.remove(id);
                }
                if (entry == null || entry[0] == null) return;
                boolean success = status == DownloadManager.STATUS_SUCCESSFUL;
                String fileName = (String) entry[2];
                if (success) fileName = renameBySniff(localUri, fileName);
                @SuppressWarnings("unchecked")
                WebView webView = ((WeakReference<WebView>) entry[0]).get();
                State state = webView == null ? null : states.get(webView);
                if (state != null && state.events != null) {
                    state.events.onDownloadComplete((String) entry[1], fileName, success);
                }
            }
        };
        IntentFilter filter = new IntentFilter(DownloadManager.ACTION_DOWNLOAD_COMPLETE);
        if (Build.VERSION.SDK_INT >= 33) {
            // sent by the system's download app: exported (Android 14 asks for the flag)
            context.registerReceiver(receiver, filter, Context.RECEIVER_EXPORTED);
        } else {
            context.registerReceiver(receiver, filter);
        }
    }

    /** A long press on an image (also an image inside a link) offers "Download image". */
    private static void imageLongPress(final WebView webView, final State state) {
        webView.setOnLongClickListener(new View.OnLongClickListener() {
            @Override
            public boolean onLongClick(View v) {
                WebView.HitTestResult hit = webView.getHitTestResult();
                if (hit == null) return false;
                if (hit.getType() == WebView.HitTestResult.IMAGE_TYPE) {
                    String src = hit.getExtra();
                    if (src == null || !(src.startsWith("http://") || src.startsWith("https://"))) return false;
                    offerImage(webView, state, src);
                    return true;
                }
                if (hit.getType() == WebView.HitTestResult.SRC_IMAGE_ANCHOR_TYPE) {
                    // the image's own address (not the link's) comes back in a message ("url")
                    Handler handler = new Handler(Looper.getMainLooper()) {
                        @Override
                        public void handleMessage(Message message) {
                            Bundle data = message.getData();
                            String src = data == null ? null : data.getString("url");
                            if (src == null && data != null) src = data.getString("src");
                            offerImage(webView, state, src);
                        }
                    };
                    webView.requestImageRef(handler.obtainMessage());
                    return true;
                }
                return false;
            }
        });
    }

    private static void offerImage(final WebView webView, final State state, final String src) {
        if (src == null || !(src.startsWith("http://") || src.startsWith("https://"))) return;
        final String name = fileName(src, null, null);
        DialogInterface.OnClickListener download = new DialogInterface.OnClickListener() {
            @Override
            public void onClick(DialogInterface dialog, int which) {
                enqueue(webView.getContext(), webView, state, src, webView.getSettings().getUserAgentString(),
                        name, mimeType(null, name), state.folder, state.notify);
            }
        };
        String[] items = {"Download image"};
        try {
            new AlertDialog.Builder(webView.getContext()).setItems(items, download).show();
        } catch (Exception e) {
            // a theme without Material's colours: the framework dialog
            try {
                new android.app.AlertDialog.Builder(webView.getContext()).setItems(items, download).show();
            } catch (Exception ignored) {
                // the screen is closing
            }
        }
    }

    /** Distinctive ad / tracker fragments matched anywhere in a request URL (a broad second layer over the host list). */
    private static final String[] AD_URL_FRAGMENTS = {"doubleclick", "popads", "popcash", "popunder", "adservice",
            "adserver", "adsystem", "adsense", "googlesyndication", "googleadservices", "googletagservices",
            "googletagmanager", "google-analytics", "/pagead", "/adsales", "zedo", "adnxs", "taboola", "outbrain",
            "/ad_", "/ads/", "/ads?", "/adx/", "=ads&", "scorecardresearch", "quantserve", "moatads", "adroll",
            "criteo", "/tracking/", "/telemetry/", "/beacon"};

    /** A broad check: an ad / tracker host, or a URL that carries a distinctive ad fragment. */
    public static boolean isAdUrl(String url) {
        if (url == null) return false;
        String host = Uri.parse(url).getHost();
        if (isAdHost(host)) return true;
        String lower = url.toLowerCase(Locale.US);
        for (int i = 0; i < AD_URL_FRAGMENTS.length; i++) {
            if (lower.contains(AD_URL_FRAGMENTS[i])) return true;
        }
        return false;
    }

    /** Whether the host or one of its parent domains is on the built-in ad and tracker list. */
    public static boolean isAdHost(String host) {
        if (host == null) return false;
        String h = host.toLowerCase(Locale.US);
        while (h.length() > 0) {
            if (adHosts.contains(h)) return true;
            int dot = h.indexOf('.');
            if (dot < 0) return false;
            h = h.substring(dot + 1);
        }
        return false;
    }

    /**
     * Blocks requests to ad and tracker hosts (Android 8+: the WebView's own client is kept and wrapped; on older
     * Android the WebView's client can't be read, so nothing is blocked there).
     */
    public static void setAdBlockEnabled(WebView webView, boolean enabled) {
        if (webView == null) return;
        State state = state(webView);
        state.adBlock = enabled;
        if (enabled && Build.VERSION.SDK_INT >= 26) {
            WebViewClient current = webView.getWebViewClient();
            if (!(current instanceof Client)) webView.setWebViewClient(new Client(current, state));
        }
    }

    /**
     * On: a page opens a new window (pop-up) only after a tap, and then in this WebView; pop-ups a script opens on its
     * own are blocked. Off: scripts may open them (they load in this WebView).
     */
    public static void setPopupBlockEnabled(WebView webView, boolean enabled) {
        if (webView == null) return;
        State state = state(webView);
        state.popupBlock = enabled;
        WebSettings settings = webView.getSettings();
        settings.setJavaScriptCanOpenWindowsAutomatically(!enabled);
        settings.setSupportMultipleWindows(enabled);
        if (enabled) chrome(webView, state);
    }

    private static void chrome(WebView webView, State state) {
        WebChromeClient current = Build.VERSION.SDK_INT >= 26 ? webView.getWebChromeClient() : null;
        if (current instanceof Chrome) return;
        if (Build.VERSION.SDK_INT < 26 && state.chrome != null) return;
        // Android 8+: the app's own client (progress, title, full screen video) keeps working inside this one
        state.chrome = new Chrome(state, current);
        webView.setWebChromeClient(state.chrome);
    }

    /**
     * Second ad-blocking layer: hides ad containers the network layer let through, by injecting a style sheet. Needs
     * JavaScript (the WebView's initializer turns it on); does nothing when it is off.
     */
    private static final String HIDING_CSS =
            ".adsbygoogle,[id^=\"google_ads\"],[id^=\"div-gpt-ad\"],[class*=\"ad-box\"],[class*=\"ad-banner\"],"
                    + "[class*=\"advert\"],[class*=\"-ads\"],[class*=\"ads-\"],ins.adsbygoogle,iframe[src*=\"doubleclick\"],"
                    + "iframe[src*=\"googlesyndication\"],iframe[id*=\"google_ads\"],.popunder,.pop-under,div[id*=\"popup-ad\"],"
                    + "div[class*=\"popup-ad\"]{display:none !important;height:0 !important;max-height:0 !important;"
                    + "pointer-events:none !important;visibility:hidden !important;}";

    private static void injectHidingCss(WebView view) {
        try {
            String js = "(function(){try{var id='sketch_adblock_css';if(document.getElementById(id))return;"
                    + "var s=document.createElement('style');s.id=id;s.type='text/css';"
                    + "s.appendChild(document.createTextNode(" + jsString(HIDING_CSS) + "));"
                    + "(document.head||document.documentElement).appendChild(s);}catch(e){}})();";
            view.evaluateJavascript(js, null);
        } catch (Exception ignored) {
            // JavaScript off, or the render process gone
        }
    }

    /** A JavaScript string literal for a value (quotes, backslashes and newlines escaped). */
    private static String jsString(String value) {
        StringBuilder out = new StringBuilder("\"");
        for (int i = 0; i < value.length(); i++) {
            char c = value.charAt(i);
            if (c == '\\' || c == '"') out.append('\\').append(c);
            else if (c == '\n') out.append("\\n");
            else if (c == '\r') out.append("\\r");
            else out.append(c);
        }
        return out.append('"').toString();
    }

    /**
     * Goes back, skipping ad / redirect pages so Back doesn't land on a bounce page (owner's v1.9.7 "redirect-free
     * back"). Returns whether it moved; false when there is no earlier real page (the caller then lets the screen
     * close).
     */
    public static boolean goBack(WebView webView) {
        if (webView == null) return false;
        WebBackForwardList list = webView.copyBackForwardList();
        int current = list.getCurrentIndex();
        int target = current - 1;
        while (target >= 0 && isAdUrl(list.getItemAtIndex(target).getUrl())) target--;
        if (target < 0) {
            if (current <= 0) return false;
            target = current - 1;   // every earlier page looked like an ad: step back one rather than get stuck
        }
        int delta = target - current;
        if (webView.canGoBackOrForward(delta)) {
            webView.goBackOrForward(delta);
            return true;
        }
        return false;
    }

    /** One WebChromeClient for file upload and pop-ups; everything else goes to the app's own client (Android 8+). */
    private static class Chrome extends WebChromeClient {

                    if (done[0]) return;
                    done[0] = true;
                    destroy.run();
                }
            }, 30000);
            WebView.WebViewTransport transport = (WebView.WebViewTransport) resultMsg.obj;
            transport.setWebView(popup);
            resultMsg.sendToTarget();

        }

    /** http(s) loads in the WebView with the page as Referer; other links (intent:, mailto:, apps) go to Android. */
    private static void openUrl(WebView view, String url, String referer) {
        if (url.startsWith("http://") || url.startsWith("https://")) {
            Map<String, String> headers = new HashMap<String, String>();
            if (referer != null && (referer.startsWith("http://") || referer.startsWith("https://"))) {
                headers.put("Referer", referer);
            }
            view.loadUrl(url, headers);
            return;
        }
        try {
            Intent intent;
            if (url.startsWith("intent:")) {
                intent = Intent.parseUri(url, Intent.URI_INTENT_SCHEME);
                // like a browser: no component, selector or flags (no permission grants) from a web page
                intent.setComponent(null);
                intent.setSelector(null);
                intent.setFlags(0);
                intent.addCategory(Intent.CATEGORY_BROWSABLE);
            } else {
                intent = new Intent(Intent.ACTION_VIEW, Uri.parse(url));
            }
            Uri data = intent.getData();
            String scheme = data == null || data.getScheme() == null ? "" : data.getScheme().toLowerCase(Locale.US);
            // the app's own files never go out through a web page's link
            if (scheme.equals("content") || scheme.equals("file") || scheme.equals("javascript")) return;
            if (!(view.getContext() instanceof Activity)) intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            view.getContext().startActivity(intent);
        } catch (Exception ignored) {
            // no app for the link
        }
    }

    /** COMPATIBILITY (Chrome's rule: http images on an https page load, scripts don't), NEVER_ALLOW or ALWAYS_ALLOW. */
    public static void setMixedContent(WebView webView, String mode) {
        if (webView == null || Build.VERSION.SDK_INT < 21) return;
        int value = WebSettings.MIXED_CONTENT_COMPATIBILITY_MODE;
        if ("NEVER_ALLOW".equals(mode)) value = WebSettings.MIXED_CONTENT_NEVER_ALLOW;
        if ("ALWAYS_ALLOW".equals(mode)) value = WebSettings.MIXED_CONTENT_ALWAYS_ALLOW;
        webView.getSettings().setMixedContentMode(value);
    }

    /** Opens Google's results for the text. */
    public static void searchGoogle(WebView webView, String query) {
        if (webView == null || query == null) return;
        try {
            webView.loadUrl("https://www.google.com/search?q=" + java.net.URLEncoder.encode(query.trim(), "UTF-8"));
        } catch (Exception ignored) {
        }
    }

    /** Puts the addresses the WebView visited (oldest first) into the list. */
    public static void history(WebView webView, ArrayList<String> list) {
        if (webView == null || list == null) return;
        list.clear();
        WebBackForwardList pages = webView.copyBackForwardList();
        for (int i = 0; i < pages.getSize(); i++) list.add(pages.getItemAtIndex(i).getUrl());
    }

    /** A user agent of your own; empty: the WebView's. */
    public static void setUserAgent(WebView webView, String userAgent) {
        if (webView == null) return;
        webView.getSettings().setUserAgentString(userAgent == null || userAgent.trim().length() == 0 ? null : userAgent.trim());
    }

    /** A desktop browser's user agent with this WebView's Chrome version. */
    public static String desktopUserAgent(String mobile) {
        String chrome = "";
        if (mobile != null) {
            int at = mobile.indexOf("Chrome/");
            if (at >= 0) {
                int end = mobile.indexOf(' ', at);
                chrome = (end < 0 ? mobile.substring(at) : mobile.substring(at, end)) + " ";
            }
        }
        return "Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) " + chrome + "Safari/537.36";
    }

    /** Desktop mode on: desktop user agent and wide view port; off: the WebView's own. Reloads the page. */
    public static void setDesktopMode(WebView webView, boolean desktop) {
        if (webView == null) return;
        WebSettings settings = webView.getSettings();
        settings.setUserAgentString(desktop ? desktopUserAgent(WebSettings.getDefaultUserAgent(webView.getContext())) : null);
        settings.setUseWideViewPort(desktop);
        settings.setLoadWithOverviewMode(desktop);
        if (webView.getUrl() != null) webView.reload();
    }

    /** Clears the cache, cookies, web storage, form data and history. */
    @SuppressWarnings("deprecation")
    public static void clearCookiesAndCache(WebView webView) {
        if (webView != null) {
            webView.clearCache(true);
            webView.clearFormData();
            webView.clearHistory();
        }
        CookieManager cookies = CookieManager.getInstance();
        if (Build.VERSION.SDK_INT >= 21) {
            cookies.removeAllCookies(null);
            cookies.flush();
        } else {
            cookies.removeAllCookie();
        }
        WebStorage.getInstance().deleteAllData();
    }

    /**
     * Lets the page's file inputs open the file picker (Android 5+). Sets the WebView's WebChromeClient (shared with
     * the pop-up block): one set by code before is replaced.
     */
    public static void enableFileUpload(Object host, WebView webView) {
        Activity activity = activity(host);
        if (activity == null || webView == null) return;
        State state = state(webView);
        state.uploads = true;
        state.activity = new WeakReference<Activity>(activity);
        chrome(webView, state);
    }

    @SuppressWarnings("deprecation")
    private static Activity activity(Object host) {
        if (host instanceof Activity) return (Activity) host;
        /*@ANDROIDX_FRAGMENT*/
        if (host instanceof android.app.Fragment) return ((android.app.Fragment) host).getActivity();
        if (host instanceof View && ((View) host).getContext() instanceof Activity) {
            return (Activity) ((View) host).getContext();
        }
        try {
            Object activity = host.getClass().getMethod("getActivity").invoke(host);
            if (activity instanceof Activity) return (Activity) activity;
        } catch (Exception ignored) {
        }
        return null;
    }

    private static Context context(Object host) {
        if (host instanceof Context) return (Context) host;
        return activity(host);
    }

    /**
     * Opens the picker from a fragment of its own, so the answer reaches it whatever the activity's
     * onActivityResult does.
     */
    @SuppressWarnings("deprecation")
    public static class Chooser extends android.app.Fragment {
        private static final String TAG = "sketch_web_chooser";
        private static ValueCallback<Uri[]> callback;
        private static Intent intent;

        public Chooser() {
        }

        static boolean open(Activity activity, Intent pick, ValueCallback<Uri[]> answer) {
            if (callback != null) callback.onReceiveValue(null);
            callback = answer;
            intent = pick;
            try {
                android.app.FragmentManager manager = activity.getFragmentManager();
                android.app.Fragment old = manager.findFragmentByTag(TAG);
                if (old != null) manager.beginTransaction().remove(old).commitAllowingStateLoss();
                manager.beginTransaction().add(new Chooser(), TAG).commitAllowingStateLoss();
                return true;
            } catch (Exception e) {
                callback = null;
                answer.onReceiveValue(null);
                return true;
            }
        }

        @Override
        public void onCreate(Bundle savedInstanceState) {
            super.onCreate(savedInstanceState);
            if (savedInstanceState != null || intent == null) {
                // the activity was recreated: the page that asked is gone
                finish(null);
                return;
            }
            try {
                startActivityForResult(Intent.createChooser(intent, null), UPLOAD_CODE);
            } catch (Exception e) {
                finish(null);
            }
            intent = null;
        }

        @Override
        public void onActivityResult(int requestCode, int resultCode, Intent data) {
            if (requestCode != UPLOAD_CODE) return;
            Uri[] uris = null;
            if (resultCode == Activity.RESULT_OK && data != null) {
                ClipData clip = data.getClipData();
                if (clip != null && clip.getItemCount() > 0) {
                    uris = new Uri[clip.getItemCount()];
                    for (int i = 0; i < uris.length; i++) uris[i] = clip.getItemAt(i).getUri();
                } else if (data.getData() != null) {
                    uris = new Uri[]{data.getData()};
                }
            }
            finish(uris);
        }

        private void finish(Uri[] uris) {
            ValueCallback<Uri[]> answer = callback;
            callback = null;
            if (answer != null) answer.onReceiveValue(uris);
            try {
                getFragmentManager().beginTransaction().remove(this).commitAllowingStateLoss();
            } catch (Exception ignored) {
            }
        }
    }
